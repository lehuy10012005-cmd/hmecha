"use client";

import Link from "next/link";
import { FormEvent, useEffect, useState } from "react";
import { useSearchParams } from "next/navigation";

export default function ForgotPasswordPage() {
  const params = useSearchParams();
  const [email, setEmail] = useState("");
  const [loading, setLoading] = useState(false);
  const [message, setMessage] = useState("");
  const [error, setError] = useState("");

  useEffect(() => {
    const emailParam = params.get("email");

    if (emailParam) {
      setEmail(emailParam);
    }
  }, [params]);

  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();

    setLoading(true);
    setMessage("");
    setError("");

    const response = await fetch("/api/auth/forgot-password-custom", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
      },
      body: JSON.stringify({ email }),
    });

    const data = await response.json();

    if (!response.ok) {
      setError(data.message || "Không gửi được email đặt lại mật khẩu.");
      setLoading(false);
      return;
    }

    setMessage(data.message);
    setLoading(false);
  }

  return (
    <main className="hmAuthPage">
      <section className="hmAuthCard">
        <Link href="/" className="hmAuthBrand">
          <b>HMECHA</b>
          <span>MEMBER CENTER</span>
        </Link>

        <p className="hmEyebrow">PASSWORD RECOVERY</p>
        <h1>Quên mật khẩu</h1>

        <p className="hmDesc">
          Nhập email đã đăng ký. HMECHA sẽ gửi link để bạn tạo mật khẩu mới.
        </p>

        {message ? <div className="hmSuccess">{message}</div> : null}
        {error ? <div className="hmError">{error}</div> : null}

        <form onSubmit={submit} className="hmForm">
          <label>
            Email tài khoản
            <input
              type="email"
              value={email}
              onChange={(event) => setEmail(event.target.value)}
              placeholder="email@example.com"
              required
            />
          </label>

          <button disabled={loading}>
            {loading ? "Đang gửi..." : "Gửi link đặt lại mật khẩu"}
          </button>
        </form>

        <p className="hmSwitch">
          Nhớ mật khẩu rồi? <Link href="/dang-nhap">Đăng nhập</Link>
        </p>
      </section>

      <style>{`
        .hmAuthPage {
          min-height: 100vh;
          display: grid;
          place-items: center;
          padding: 28px;
          background:
            radial-gradient(circle at 16% 12%, rgba(124,77,255,.3), transparent 34%),
            radial-gradient(circle at 82% 16%, rgba(0,229,255,.2), transparent 32%),
            linear-gradient(180deg, #050816 0%, #0b1026 100%);
          color: #ffffff;
        }

        .hmAuthCard {
          width: min(520px, 100%);
          box-sizing: border-box;
          padding: 38px;
          border-radius: 28px;
          background: rgba(11, 16, 38, .96);
          border: 1px solid rgba(0,229,255,.24);
          box-shadow: 0 28px 90px rgba(0,0,0,.45);
        }

        .hmAuthBrand {
          display: inline-flex;
          flex-direction: column;
          margin-bottom: 28px;
          text-decoration: none;
        }

        .hmAuthBrand b {
          color: #00e5ff;
          font-size: 27px;
          letter-spacing: 4px;
          font-weight: 950;
        }

        .hmAuthBrand span {
          margin-top: 6px;
          color: #9eadd3;
          font-size: 12px;
          font-weight: 850;
          letter-spacing: 1px;
        }

        .hmEyebrow {
          margin: 0 0 12px;
          color: #00e5ff;
          font-size: 12px;
          font-weight: 950;
          letter-spacing: 2px;
        }

        .hmAuthCard h1 {
          margin: 0 0 12px;
          color: #ffffff;
          font-size: 38px;
          line-height: 1.1;
        }

        .hmDesc {
          margin: 0 0 24px;
          color: #cbd5e1;
          font-size: 16px;
          line-height: 1.6;
        }

        .hmSuccess,
        .hmError {
          margin: 0 0 18px;
          padding: 14px 15px;
          border-radius: 14px;
          font-size: 14px;
          line-height: 1.55;
        }

        .hmSuccess {
          color: #63f1ad;
          background: rgba(45,205,124,.12);
          border: 1px solid rgba(45,205,124,.28);
        }

        .hmError {
          color: #ffb4c0;
          background: rgba(255,70,96,.13);
          border: 1px solid rgba(255,70,96,.3);
        }

        .hmForm {
          display: grid;
          gap: 18px;
        }

        .hmForm label {
          color: #e5edff;
          font-size: 14px;
          font-weight: 850;
        }

        .hmForm input {
          display: block;
          width: 100%;
          box-sizing: border-box;
          margin-top: 8px;
          padding: 15px 16px;
          border: 1px solid rgba(255,255,255,.16);
          border-radius: 14px;
          outline: none;
          background: rgba(255,255,255,.075);
          color: #ffffff;
          font-size: 15px;
        }

        .hmForm input:focus {
          border-color: #00e5ff;
          box-shadow: 0 0 0 3px rgba(0,229,255,.13);
        }

        .hmForm button {
          margin-top: 8px;
          min-height: 54px;
          border: none;
          border-radius: 15px;
          background: linear-gradient(135deg,#7c4dff,#00e5ff);
          color: #050816;
          cursor: pointer;
          font-size: 16px;
          font-weight: 950;
          box-shadow: 0 18px 34px rgba(0,229,255,.18);
        }

        .hmForm button:disabled {
          opacity: .65;
          cursor: not-allowed;
        }

        .hmSwitch {
          margin: 24px 0 0;
          text-align: center;
          color: #cbd5e1;
        }

        .hmSwitch a {
          color: #00e5ff;
          font-weight: 950;
          text-decoration: none;
        }
      `}</style>
    </main>
  );
}
